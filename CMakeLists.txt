cmake_minimum_required(VERSION 3.16)
project(tmnt_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(TMNT_TOOLS rominfo disasm flow hwscan vectors entrydump callscan callgraph cfg lift exec)
set(TMNT_TEST_EXECUTABLES)
foreach(tool IN LISTS TMNT_TOOLS)
  add_executable(tmnt_${tool} src/${tool}.cpp)
  if(MSVC)
    target_compile_options(tmnt_${tool} PRIVATE /W4)
  else()
    target_compile_options(tmnt_${tool} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  list(APPEND TMNT_TEST_EXECUTABLES "${tool}=$<TARGET_FILE:tmnt_${tool}>")
  list(APPEND TMNT_CHECK_TARGETS tmnt_${tool})
endforeach()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME tools_smoke COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py" ${TMNT_TEST_EXECUTABLES})
  set_tests_properties(tools_smoke PROPERTIES TIMEOUT 30)
endif()

if(WIN32)
  add_executable(tmnt_native WIN32 src/native_windows.cpp)
  target_link_libraries(tmnt_native PRIVATE comdlg32 user32 gdi32)
endif()

add_executable(tmnt_cpuprobe src/cpuprobe.cpp)
if(MSVC)
  target_compile_options(tmnt_cpuprobe PRIVATE /W4)
else()
  target_compile_options(tmnt_cpuprobe PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_test(NAME cpu_probe_synthetic COMMAND "$<TARGET_FILE:tmnt_cpuprobe>" --self-test)
  add_executable(tmnt_probe_branch_tests tests/cpu_probe_branches.cpp)
  target_include_directories(tmnt_probe_branch_tests PRIVATE src)
  add_test(NAME cpu_probe_branches COMMAND "$<TARGET_FILE:tmnt_probe_branch_tests>")
  set_tests_properties(cpu_probe_branches PROPERTIES TIMEOUT 10)
  add_executable(tmnt_probe_logic_tests tests/cpu_probe_logic.cpp)
  target_include_directories(tmnt_probe_logic_tests PRIVATE src)
  add_test(NAME cpu_probe_logic COMMAND "$<TARGET_FILE:tmnt_probe_logic_tests>")
  set_tests_properties(cpu_probe_logic PROPERTIES TIMEOUT 10)
  add_executable(tmnt_probe_chain_tests tests/cpu_probe_chain.cpp)
  target_include_directories(tmnt_probe_chain_tests PRIVATE src)
  add_test(NAME cpu_probe_chain COMMAND "$<TARGET_FILE:tmnt_probe_chain_tests>")
  set_tests_properties(cpu_probe_chain PROPERTIES TIMEOUT 10)
  add_executable(tmnt_probe_arithmetic_tests tests/cpu_probe_arithmetic.cpp)
  target_include_directories(tmnt_probe_arithmetic_tests PRIVATE src)
  add_test(NAME cpu_probe_arithmetic COMMAND "$<TARGET_FILE:tmnt_probe_arithmetic_tests>")
  set_tests_properties(cpu_probe_arithmetic PROPERTIES TIMEOUT 30)
  add_executable(tmnt_probe_memory_tests tests/cpu_probe_memory.cpp)
  target_include_directories(tmnt_probe_memory_tests PRIVATE src)
  add_test(NAME cpu_probe_memory COMMAND "$<TARGET_FILE:tmnt_probe_memory_tests>")
  set_tests_properties(cpu_probe_memory PROPERTIES TIMEOUT 10)
  add_custom_target(check
    COMMAND "${CMAKE_CTEST_COMMAND}" -C "$<CONFIG>" --output-on-failure --output-junit ctest-results.xml
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS ${TMNT_CHECK_TARGETS} tmnt_cpuprobe tmnt_probe_branch_tests tmnt_probe_logic_tests tmnt_probe_chain_tests tmnt_probe_arithmetic_tests tmnt_probe_memory_tests
    USES_TERMINAL)
  if(TARGET tmnt_native)
    add_dependencies(check tmnt_native)
  endif()
endif()

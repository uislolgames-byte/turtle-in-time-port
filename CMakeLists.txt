cmake_minimum_required(VERSION 3.16)
project(tmnt_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(TMNT_TOOLS rominfo disasm flow hwscan vectors entrydump callscan callgraph cfg lift exec)
set(TMNT_TEST_EXECUTABLES)
foreach(tool IN LISTS TMNT_TOOLS)
  add_executable(tmnt_${tool} src/${tool}.cpp)
  if(MSVC)
    target_compile_options(tmnt_${tool} PRIVATE /W4)
  else()
    target_compile_options(tmnt_${tool} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  list(APPEND TMNT_TEST_EXECUTABLES "${tool}=$<TARGET_FILE:tmnt_${tool}>")
  list(APPEND TMNT_CHECK_TARGETS tmnt_${tool})
endforeach()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME tools_smoke COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py" ${TMNT_TEST_EXECUTABLES})
  set_tests_properties(tools_smoke PROPERTIES TIMEOUT 30)
endif()

if(WIN32)
  add_executable(tmnt_native WIN32 src/native_windows.cpp)

    set(fixture_bin "${CMAKE_CURRENT_BINARY_DIR}/recompiler_${fixture}.bin")
    set(fixture_cpp "${CMAKE_CURRENT_BINARY_DIR}/recompiler_${fixture}.cpp")
    set(fixture_options)
    if(fixture MATCHES "^irq_")
      set(fixture_options --interrupts)
    elseif(fixture MATCHES "^bus_")
      set(fixture_options --bus-timing)
    endif()
    add_custom_command(OUTPUT "${fixture_cpp}" "${fixture_bin}"
      COMMAND "$<TARGET_FILE:tmnt_recompiler_fixtures>" "${fixture}" "${fixture_bin}"
      COMMAND "$<TARGET_FILE:tmnt_recompile>" "${fixture_bin}" "${fixture_cpp}" ${fixture_options}
      DEPENDS tmnt_recompiler_fixtures tmnt_recompile tests/recompiler_fixtures.hpp
      VERBATIM)

      set(fixture_test tests/recompiler_stack_timing.cpp)
    elseif(fixture MATCHES "^bus_move_")
      set(fixture_test tests/recompiler_transfer.cpp)
    elseif(fixture MATCHES "^bus_")
      set(fixture_test tests/recompiler_bus_timing.cpp)
    elseif(fixture STREQUAL "system")
      set(fixture_test tests/recompiler_system.cpp)
    elseif(fixture STREQUAL "memsel")
      set(fixture_test tests/recompiler_memsel.cpp)
    elseif(fixture MATCHES "^move_")
      set(fixture_test tests/recompiler_transfer.cpp)
    elseif(fixture MATCHES "^irq_")
      set(fixture_test tests/recompiler_interrupts.cpp)
    elseif(fixture MATCHES "^dma_")
      set(fixture_test tests/recompiler_dma.cpp)
    else()
      set(fixture_test tests/recompiler_equivalence.cpp)
    endif()
    add_executable(tmnt_recompiled_${fixture} "${fixture_test}" "${fixture_cpp}")
    target_include_directories(tmnt_recompiled_${fixture} PRIVATE recompiler)
    add_test(NAME recompiled_${fixture} COMMAND tmnt_recompiled_${fixture} "${fixture}")
    if(fixture MATCHES "^irq_")
      set_tests_properties(recompiled_${fixture} PROPERTIES TIMEOUT 40)
    else()
      set_tests_properties(recompiled_${fixture} PROPERTIES TIMEOUT 20)
    endif()
    add_dependencies(check tmnt_recompiled_${fixture})
  endforeach()
endif()

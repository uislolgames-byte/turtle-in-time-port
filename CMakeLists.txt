cmake_minimum_required(VERSION 3.16)
project(tmnt_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(tool IN ITEMS rominfo disasm flow hwscan)
  add_executable(tmnt_${tool} src/${tool}.cpp)
  if(MSVC)
    target_compile_options(tmnt_${tool} PRIVATE /W4)
  else()
    target_compile_options(tmnt_${tool} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME tools_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py" "$<TARGET_FILE:tmnt_rominfo>" "$<TARGET_FILE:tmnt_disasm>" "$<TARGET_FILE:tmnt_flow>" "$<TARGET_FILE:tmnt_hwscan>")
  set_tests_properties(tools_smoke PROPERTIES TIMEOUT 30)
endif()
